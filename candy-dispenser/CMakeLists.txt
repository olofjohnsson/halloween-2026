cmake_minimum_required(VERSION 3.20.0)
# Wi-Fi/server settings live in a git-ignored file (see wifi_secrets.conf.example)
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/wifi_secrets.conf)
  list(APPEND EXTRA_CONF_FILE ${CMAKE_CURRENT_SOURCE_DIR}/wifi_secrets.conf)
endif()
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(halloween_candy_dispenser)
target_sources(app PRIVATE src/main.c src/report.c ../common/fota.c)
target_include_directories(app PRIVATE ../common)
if(CONFIG_HALLOWEEN_DFPLAYER)
  target_sources(app PRIVATE ../common/dfplayer.c)
endif()
